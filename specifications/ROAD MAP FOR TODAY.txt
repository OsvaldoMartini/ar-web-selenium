  public final class PageDiagnosticDumper {

      private static final String JS_OUTER_HTML =
              "return document.documentElement.outerHTML;";

      private static final String JS_META =
              "return JSON.stringify({" +
              "url: location.href," +
              "userAgent: navigator.userAgent," +
              "devicePixelRatio: window.devicePixelRatio," +
              "innerWidth: window.innerWidth, innerHeight: window.innerHeight," +
              "scrollX: window.scrollX, scrollY: window.scrollY," +
              "documentWidth: document.documentElement.scrollWidth," +
              "documentHeight: document.documentElement.scrollHeight," +
              "readyState: document.readyState," +
              "timestamp: new Date().toISOString()" +
              "});";

      private static final String JS_SHADOW_SURVEY =
              "function walk(n, path, out){" +
              "  if(n.shadowRoot){out.push({host:path, hostTag:n.tagName, mode:n.shadowRoot.mode});" +
              "    for(const c of n.shadowRoot.children) walk(c, path+'>'+c.tagName, out);}" +
              "  for(const c of (n.children||[])) walk(c, path+'>'+c.tagName, out);" +
              "} const out=[]; walk(document.documentElement,'html',out); return JSON.stringify(out);";

      private static final String JS_RECTS =
              "const xps = arguments[0]; const out = [];" +
              "for (const xp of xps) {" +
              "  let r=null, node=null, tag=null, id='', cls='';" +
              "  try { const res = document.evaluate(xp, document, null, XPathResult.FIRST_ORDERED_NODE_TYPE, null);" +
              "    node = res.singleNodeValue;" +
              "    if (node) { tag = node.tagName; id = node.id||''; cls = (node.className &&
  node.className.baseVal!==undefined?node.className.baseVal:(node.className||''));" +
              "      const b = node.getBoundingClientRect();" +
              "      r = {x:b.x,y:b.y,width:b.width,height:b.height,top:b.top,left:b.left,bottom:b.bottom,right:b.right}; }" +
              "  } catch(e) {}" +
              "  out.push({xpath:xp, found:!!node, tag:tag, id:id, class:cls, rect:r});" +
              "} return JSON.stringify(out);";

      private PageDiagnosticDumper() {}

      /** Call at clone-mode activation. Captures page-wide snapshot. */
      public static void dumpPage(WebDriver driver, String outDir, String prefix) {
          try {
              Files.createDirectories(Paths.get(outDir));
              JavascriptExecutor js = (JavascriptExecutor) driver;

              // 1. Selenium-side page source
              write(outDir, prefix + ".html", driver.getPageSource());

              // 2. Live outerHTML
              write(outDir, prefix + "-live.html", (String) js.executeScript(JS_OUTER_HTML));

              // 3. Viewport / DPR / URL meta
              write(outDir, prefix + "-meta.json", (String) js.executeScript(JS_META));

              // 4. Shadow DOM survey
              write(outDir, prefix + "-shadow.json", (String) js.executeScript(JS_SHADOW_SURVEY));
          } catch (Exception ex) {
              System.err.println("[PageDiagnosticDumper] dumpPage failed: " + ex.getMessage());
          }
      }

      /** Call after DTO is written. Captures bounding rects for every xPath in the DTO. */
      public static void dumpRects(WebDriver driver, List<String> xPaths, String outDir, String prefix) {
          try {
              Files.createDirectories(Paths.get(outDir));
              JavascriptExecutor js = (JavascriptExecutor) driver;
              String json = (String) js.executeScript(JS_RECTS, xPaths);
              write(outDir, prefix + "-rects.json", json);
          } catch (Exception ex) {
              System.err.println("[PageDiagnosticDumper] dumpRects failed: " + ex.getMessage());
          }
      }

      private static void write(String dir, String name, String content) throws IOException {
          Path p = Paths.get(dir, name);
          Files.writeString(p, content == null ? "" : content, StandardCharsets.UTF_8);
      }
  }

  Coordinate alignment — the tricky part

  Three spaces must agree before correlation:
  - DTO coordinates — CSS px, already (your values are 684.50, 448.00 on a page that is clearly higher-res, so not device px).
  - getBoundingClientRect() — CSS px, viewport-relative.
  - Screenshot pixels — device px = CSS × DPR.

  So the correlator converts: ocr_x_css = ocr_x_px / dpr, ocr_y_css = ocr_y_px / dpr, plus scroll offsets if the screenshot is a stitched full-page capture. The meta
  JSON from Roadmap 1 carries all three values — that's why it matters.

  Correlation confidence tiers (for Roadmap 3 downstream)

  - EXACT_CONTAIN — OCR word center is inside DTO bbox → very strong.
  - OVERLAP — OCR word bbox intersects DTO bbox but center is outside → strong.
  - PROXIMITY — nearest OCR word within ≤30 px of DTO center → medium.
  - NONE — no OCR text nearby (icon-only button, input without label) → weak; fall back to color-button OCR label.

      homebanking_id    VARCHAR(64)  NOT NULL,
      page_url_hash     CHAR(64)     NOT NULL,   -- SHA-256 of normalized URL
      page_title        VARCHAR(255),
      x_path            TEXT         NOT NULL,

      attrib_id         VARCHAR(255),
      attrib_name       VARCHAR(255),

  -- (someText changed, xpath no longer resolves, etc.), write a row here.
  CREATE TABLE element_locator_rename (
      id                BIGINT PRIMARY KEY AUTO_INCREMENT,
      locator_id        BIGINT       NOT NULL,
      observed_at       TIMESTAMP    DEFAULT CURRENT_TIMESTAMP,
      change_type       ENUM('TEXT','XPATH','ATTRIB_ID','ATTRIB_NAME','COORDS','OTHER') NOT NULL,
      some_text_new     VARCHAR(512),
      defined_name_new  VARCHAR(255),
      x_path_new        TEXT,
      attrib_id_new     VARCHAR(255),
      attrib_name_new   VARCHAR(255),
      coords_new        VARCHAR(64),
      ocr_text_new      VARCHAR(512),
      match_confidence  DECIMAL(5,2),      -- 0..100 from recovery algorithm
      recovery_strategy VARCHAR(64),       -- 'TEXT_EXACT','TEXT_FUZZY','OCR','COORDS','CSS','COMBINED'
      FOREIGN KEY (locator_id) REFERENCES element_locator(id) ON DELETE CASCADE,
      INDEX ix_locator (locator_id),
      INDEX ix_observed (observed_at)
  );

  Recovery algorithm (when xPath fails at bot-run time)

  findOrRecover(locator):
    1. Try locator.x_path (and custom_x_path). If resolves + text similar → MATCH
    2. Try css_selector. If resolves + text similar → MATCH (log xpath drift)
    3. Try attrib_id / attrib_name. If resolves → MATCH (log xpath drift)
    4. Text-based:
         scan all interactive elements on page, score each by:
           - fuzzy(element.text, some_text_original)   [Levenshtein / token-set]
           - fuzzy(element.text, last_known some_text_new)
         keep top-3; if best ≥ threshold → MATCH
    5. OCR-based (Roadmap 2):
         OCR page, fuzzy-match ocr_text_original against OCR words
         project hit back to DOM via OcrDomCorrelator
    6. Coordinate-based (weakest):
         at same viewport size, pick element under coords_original (±20px)
    On any successful MATCH with drift detected → INSERT into element_locator_rename
